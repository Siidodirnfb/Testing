local function Corner(obj, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius)
    c.Parent = obj
end

local function Stroke(obj, color, thickness)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 1
    s.Parent = obj
end

local function Tween(obj, props, time)
    TweenService:Create(obj, TweenInfo.new(time or .18, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props):Play()
end

local function Text(parent, value, size, pos, font)

    t.Font = font or Enum.Font.Gotham
    t.TextSize = 12
    t.TextColor3 = TEXT
    t.TextXAlignment = Enum.TextXAlignment.Left
    t.Parent = parent
    return t
end

local function Tab(name, icon, index, page)
    local b = Instance.new("TextButton")

    b.BackgroundColor3 = SIDE
    b.Text = icon .. "   " .. name
    b.TextColor3 = MUTED
    b.Font = Enum.Font.GothamBold
    b.TextSize = 10
    b.TextXAlignment = Enum.TextXAlignment.Left
    b.BorderSizePixel = 0
    b.AutoButtonColor = false
    b.Parent = Sidebar
    Corner(b, 7)
    
    if index == 1 then
        b.BackgroundColor3 = BLUE2
        b.TextColor3 = TEXT
        b:SetAttribute("Active", true)
        page.Visible = true
    end

    Buttons[index] = b

    b.MouseButton1Click:Connect(function()
        for _, x in pairs(Buttons) do
            x:SetAttribute("Active", false)
            Tween(x, { BackgroundColor3 = SIDE, TextColor3 = MUTED })
        end
        for _, p in pairs(Pages) do
            p.Visible = false
        end
        b:SetAttribute("Active", true)
        Tween(b, { BackgroundColor3 = BLUE2, TextColor3 = TEXT })
        page.Visible = true
    end)
    return b
end

local function Toggle(page, name, y, default, callback)
    local holder = Instance.new("Frame")
    holder.Size = UDim2.new(1, -10, 0, 44)
    holder.Position = UDim2.fromOffset(0, y)
    holder.BackgroundColor3 = CARD
    holder.BorderSizePixel = 0
    holder.Parent = page
    Corner(holder, 8)

    Text(holder, name, UDim2.new(1, -70, 1, 0), UDim2.fromOffset(12, 0), Enum.Font.GothamMedium).TextSize = 11

    local button = Instance.new("TextButton")
    button.Size = UDim2.fromOffset(40, 20)
    button.Position = UDim2.new(1, -50, .5, -10)
    button.BackgroundColor3 = Color3.fromRGB(45,45,53)
    button.Text = ""
    button.BorderSizePixel = 0
    button.Parent = holder
    Corner(button, 20)

    local dot = Instance.new("Frame")
    dot.Size = UDim2.fromOffset(14, 14)
    dot.Position = UDim2.fromOffset(3, 3)
    dot.BackgroundColor3 = MUTED
    dot.BorderSizePixel = 0
    dot.Parent = button
    Corner(dot, 20)

    local enabled = default or false
    local function Update()
        if enabled then
            Tween(button, { BackgroundColor3 = BLUE })
            Tween(dot, { Position = UDim2.fromOffset(23, 3), BackgroundColor3 = Color3.new(1,1,1) })
        else
            Tween(button, { BackgroundColor3 = Color3.fromRGB(45,45,53) })
            Tween(dot, { Position = UDim2.fromOffset(3, 3), BackgroundColor3 = MUTED })
        end
        if callback then callback(enabled) end
    end

    button.MouseButton1Click:Connect(function()
        enabled = not enabled
        Update()
    end)
    Update()
end

local function Slider(page, name, min, max, default, y, callback)
    local holder = Instance.new("Frame")
    holder.Size = UDim2.new(1, -10, 0, 56)
    holder.Position = UDim2.fromOffset(0, y)
    holder.BackgroundColor3 = CARD
    holder.BorderSizePixel = 0
    holder.Parent = page
    Corner(holder, 8)

    local title = Text(holder, name, UDim2.new(1, -70, 0, 20), UDim2.fromOffset(12, 8), Enum.Font.GothamMedium)
    title.TextSize = 11

    local valLabel = Text(holder, tostring(default), UDim2.new(0, 50, 0, 20), UDim2.new(1, -60, 0, 8), Enum.Font.GothamBold)
    valLabel.TextSize = 11
    valLabel.TextXAlignment = Enum.TextXAlignment.Right

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -24, 0, 6)
    track.Position = UDim2.fromOffset(12, 36)
    track.BackgroundColor3 = Color3.fromRGB(40, 40, 48)
    track.Parent = holder
    Corner(track, 3)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.fromScale((default - min) / (max - min), 1)
    fill.BackgroundColor3 = BLUE
    fill.Parent = track
    Corner(fill, 3)

    local dragging = false
    local function UpdateInput(input)
        local posX = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + ((max - min) * posX))
        fill.Size = UDim2.fromScale(posX, 1)
        valLabel.Text = tostring(val)
        if callback then callback(val) end
    end

    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            UpdateInput(input)
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            UpdateInput(input)
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local ESPCache = {}

local function ClearESP(p)
    if ESPCache[p] then
        if ESPCache[p].Box then ESPCache[p].Box:Destroy() end
        if ESPCache[p].HealthOutline then ESPCache[p].HealthOutline:Destroy() end
        ESPCache[p] = nil
    end
    if p.Character then
        local hl = p.Character:FindFirstChild("SENSE_Highlight")
        if hl then hl:Destroy() end
    end
end

Players.PlayerRemoving:Connect(ClearESP)

-- Точный расчет границ персонажа в 3D
local function GetCharacterBoundingBox(character)
    local cframe, size = character:GetBoundingBox()
    local halfSize = size / 2

    local corners = {
        cframe * Vector3.new(-halfSize.X,  halfSize.Y, -halfSize.Z),
        cframe * Vector3.new( halfSize.X,  halfSize.Y, -halfSize.Z),
        cframe * Vector3.new(-halfSize.X, -halfSize.Y, -halfSize.Z),
        cframe * Vector3.new( halfSize.X, -halfSize.Y, -halfSize.Z),
        cframe * Vector3.new(-halfSize.X,  halfSize.Y,  halfSize.Z),
        cframe * Vector3.new( halfSize.X,  halfSize.Y,  halfSize.Z),
        cframe * Vector3.new(-halfSize.X, -halfSize.Y,  halfSize.Z),
        cframe * Vector3.new( halfSize.X, -halfSize.Y,  halfSize.Z),
    }

    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local anyOnScreen = false

    for _, corner in ipairs(corners) do
        local screenPos, onScreen = Camera:WorldToViewportPoint(corner)
        if onScreen then anyOnScreen = true end
        
        minX = math.min(minX, screenPos.X)
        minY = math.min(minY, screenPos.Y)
        maxX = math.max(maxX, screenPos.X)
        maxY = math.max(maxY, screenPos.Y)
    end

    if not anyOnScreen then return nil end

    return {
        X = minX,
        Y = minY,
        Width = maxX - minX,
        Height = maxY - minY
    }
end

local function GetClosestPlayerToCharacter()
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myHRP = myChar.HumanoidRootPart

    local target = nil
    local shortestWorldDist = math.huge
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            local enemyHRP = p.Character.HumanoidRootPart
            local screenPos, onScreen = Camera:WorldToViewportPoint(enemyHRP.Position)

            if onScreen then
                local distToCenter = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                
                if not Config.EnableFOV or distToCenter <= Config.FOVSize then
                    local worldDist = (myHRP.Position - enemyHRP.Position).Magnitude
                    
                    if worldDist < shortestWorldDist then
                        shortestWorldDist = worldDist
                        target = p.Character
                    end
                end
            end
        end
    end
    return target
end

RunService.RenderStepped:Connect(function()
    if Config.Aimbot then
        local target = GetClosestPlayerToCharacter()
        if target and target:FindFirstChild("Head") then
            Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, target.Head.Position)
        end
    end

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local char = p.Character
            local hum = char and char:FindFirstChild("Humanoid")
            local hrp = char and char:FindFirstChild("HumanoidRootPart")

            if Config.ESP and char and hum and hrp and hum.Health > 0 then
                if not ESPCache[p] then
                    ESPCache[p] = CreateESPComponents()
                end

                local components = ESPCache[p]
                local bbox = GetCharacterBoundingBox(char)

                if bbox then
                    -- Настройка ESP Box
                    if Config.ESPBox then
                        components.Box.Size = UDim2.fromOffset(bbox.Width, bbox.Height)
                        components.Box.Position = UDim2.fromOffset(bbox.X, bbox.Y)
                        
                        local boxStroke = components.Box:FindFirstChildOfClass("UIStroke")
                        if boxStroke then boxStroke.Color = Config.ESPColor end
                        
                        components.Box.Visible = true
                    else
                        components.Box.Visible = false
                    end

                    -- Настройка Health Bar (привязан слева от рамки)
                    if Config.ESPHealth then
                        local healthPercent = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                        local barWidth = 3

                        components.HealthOutline.Size = UDim2.fromOffset(barWidth + 2, bbox.Height)
                        components.HealthOutline.Position = UDim2.fromOffset(bbox.X - barWidth - 5, bbox.Y)
                        
                        components.HealthFill.Size = UDim2.new(1, -2, healthPercent, -2)
                        components.HealthFill.Position = UDim2.new(0, 1, 1 - healthPercent, 1)
                        components.HealthFill.BackgroundColor3 = Color3.fromRGB(255, 0, 0):Lerp(Color3.fromRGB(50, 220, 90), healthPercent)

                        components.HealthOutline.Visible = true
                    else
                        components.HealthOutline.Visible = false
                    end

                    -- Настройка Highlight
                    if Config.ESPHighlight then
                        local highlight = char:FindFirstChild("SENSE_Highlight")
                        if not highlight then
                            highlight = Instance.new("Highlight")
                            highlight.Name = "SENSE_Highlight"
                            highlight.Parent = char
                        end
                        highlight.FillColor = Config.ESPColor
                        highlight.OutlineColor = Config.ESPColor
                        highlight.FillTransparency = 0.5
                        highlight.Enabled = true
                    else
                        local highlight = char:FindFirstChild("SENSE_Highlight")
                        if highlight then highlight:Destroy() end
                    end
                else
                    components.Box.Visible = false
                    components.HealthOutline.Visible = false
                    local highlight = char:FindFirstChild("SENSE_Highlight")
                    if highlight then highlight:Destroy() end
                end
            else
                ClearESP(p)
            end
        end
    end
end)

local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    if Config.SilentAim and (method == "FindPartOnRayWithIgnoreList" or method == "Raycast") then
        local target = GetClosestPlayerToCharacter()
        if target and target:FindFirstChild("Head") then
            if method == "Raycast" then
                args[2] = (target.Head.Position - args[1]).Unit * 1000
            end
            return oldNamecall(self, unpack(args))
        end
    end
    return oldNamecall(self, ...)
end)
