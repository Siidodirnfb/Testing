-- ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ
local function Corner(obj, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius)
    c.Parent = obj
end

local function Stroke(obj, color, thickness)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 1
    s.Parent = obj
end

local function Tween(obj, props, time)
    TweenService:Create(obj, TweenInfo.new(time or .18, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props):Play()
end

local function Text(parent, value, size, pos, font)
    local t = Instance.new("TextLabel")

    t.TextSize = 12
    t.TextColor3 = TEXT
    t.TextXAlignment = Enum.TextXAlignment.Left
    t.Parent = parent
    return t
end

local function Tab(name, icon, index, page)
    local b = Instance.new("TextButton")
    b.Size = UDim2.fromOffset(145, 38)
    b.Position = UDim2.fromOffset(10, 80 + ((index - 1) * 44))
    b.BackgroundColor3 = SIDE
    b.Text = "   " .. icon .. "   " .. name
    b.TextColor3 = MUTED
    b.Font = Enum.Font.GothamBold
    b.TextSize = 11
    b.TextXAlignment = Enum.TextXAlignment.Left
    b.BorderSizePixel = 0
    b.AutoButtonColor = false
    b.Parent = Sidebar
    Corner(b, 7)
    Buttons[index] = b

    b.MouseEnter:Connect(function()
        if not b:GetAttribute("Active") then
            Tween(b, { BackgroundColor3 = CARD2, TextColor3 = TEXT })
        end
    end)

    b.MouseLeave:Connect(function()
        if not b:GetAttribute("Active") then
            Tween(b, { BackgroundColor3 = SIDE, TextColor3 = MUTED })
        end
    end)

    b.MouseButton1Click:Connect(function()
        for _, x in pairs(Buttons) do
            x:SetAttribute("Active", false)
            Tween(x, { BackgroundColor3 = SIDE, TextColor3 = MUTED })
        end
        for _, p in pairs(Pages) do
            p.Visible = false
        end
        b:SetAttribute("Active", true)
        Tween(b, { BackgroundColor3 = BLUE2, TextColor3 = TEXT })
        page.Visible = true
    end)

    if index == 1 then
        b:SetAttribute("Active", true)
        b.BackgroundColor3 = BLUE2
        b.TextColor3 = TEXT
        page.Visible = true
    end

    return b
end

    button.BackgroundColor3 = Color3.fromRGB(45, 45, 53)
    button.Text = ""
    button.BorderSizePixel = 0
    button.Parent = holder
    Corner(button, 20)

    local dot = Instance.new("Frame")
    dot.Size = UDim2.fromOffset(15, 15)
    dot.Position = UDim2.fromOffset(3, 3)
    dot.BackgroundColor3 = MUTED
    dot.BorderSizePixel = 0
    dot.Parent = button
    Corner(dot, 20)

    local enabled = default or false
    local cb = callback or function() end

    local function Update()
        if enabled then
            Tween(button, { BackgroundColor3 = BLUE })
            Tween(dot, { Position = UDim2.fromOffset(22, 3), BackgroundColor3 = Color3.new(1, 1, 1) })
        else
            Tween(button, { BackgroundColor3 = Color3.fromRGB(45, 45, 53) })
            Tween(dot, { Position = UDim2.fromOffset(3, 3), BackgroundColor3 = MUTED })
        end
        cb(enabled)
    end

    button.MouseButton1Click:Connect(function()
        enabled = not enabled
        Update()
    end)
    
    Update()
    return function() return enabled end, function(v) enabled = v; Update() end
end

-- КОМПОНЕНТ SLIDER (С ПОДДЕРЖКОЙ СЕНСОРА И МЫШИ)
local function Slider(page, name, min, max, default, y, callback)
    local holder = Instance.new("Frame")
    holder.Size = UDim2.new(1, -10, 0, 58)
    holder.Position = UDim2.fromOffset(0, y)
    holder.BackgroundColor3 = CARD
    holder.BorderSizePixel = 0
    holder.Parent = page
    Corner(holder, 8)

    local title = Text(holder, name, UDim2.new(0.6, 0, 0, 25), UDim2.fromOffset(13, 5), Enum.Font.GothamMedium)
    title.TextSize = 11

    local valLabel = Text(holder, tostring(default), UDim2.new(0.3, 0, 0, 25), UDim2.new(0.7, -13, 0, 5), Enum.Font.GothamBold)
    valLabel.TextSize = 11
    valLabel.TextXAlignment = Enum.TextXAlignment.Right
    valLabel.TextColor3 = BLUE

    local sliderBg = Instance.new("Frame")
    sliderBg.Size = UDim2.new(1, -26, 0, 8)
    sliderBg.Position = UDim2.fromOffset(13, 36)
    sliderBg.BackgroundColor3 = Color3.fromRGB(35, 42, 58)
    sliderBg.BorderSizePixel = 0
    sliderBg.Parent = holder
    Corner(sliderBg, 4)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = BLUE
    fill.BorderSizePixel = 0
    fill.Parent = sliderBg
    Corner(fill, 4)

    local val = default
    local sliding = false

    local function SetValue(inputVal)
        local relX = math.clamp((inputVal - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
        val = math.floor(min + (max - min) * relX)
        fill.Size = UDim2.new(relX, 0, 1, 0)
        valLabel.Text = tostring(val)
        if callback then callback(val) end
    end

    sliderBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = true
            SetValue(input.Position.X)
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            SetValue(input.Position.X)
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = false
        end
    end)
end

local function ColorPicker(page, name, defaultColor, y, callback)
    local holder = Instance.new("Frame")
    holder.Size = UDim2.new(1, -10, 0, 140)
    holder.Position = UDim2.fromOffset(0, y)
    holder.BackgroundColor3 = CARD
    holder.BorderSizePixel = 0
    holder.Parent = page
    Corner(holder, 8)

    Text(holder, name, UDim2.new(0.6, 0, 0, 30), UDim2.fromOffset(13, 5), Enum.Font.GothamMedium).TextSize = 11

    preview.Size = UDim2.fromOffset(30, 20)
    preview.Position = UDim2.new(1, -43, 0, 8)
    preview.BackgroundColor3 = defaultColor
    preview.BorderSizePixel = 0
    preview.Parent = holder
    Corner(preview, 4)
    Stroke(preview, BORDER, 1)

    local r, g, b = defaultColor.R, defaultColor.G, defaultColor.B

    local function CreateColorSlider(sliderY, labelText, defaultVal, onChange)
        local sLabel = Text(holder, labelText, UDim2.fromOffset(20, 20), UDim2.fromOffset(13, sliderY), Enum.Font.GothamBold)
        sLabel.TextSize = 10
        sLabel.TextColor3 = MUTED

        bg.Size = UDim2.new(1, -65, 0, 6)
        bg.Position = UDim2.fromOffset(40, sliderY + 7)
        bg.BackgroundColor3 = Color3.fromRGB(35, 42, 58)
        bg.BorderSizePixel = 0
        bg.Parent = holder
        Corner(bg, 3)

        local f = Instance.new("Frame")
        f.Size = UDim2.new(defaultVal, 0, 1, 0)
        f.BackgroundColor3 = BLUE
        f.BorderSizePixel = 0
        f.Parent = bg
        Corner(f, 3)

        local isDragging = false
        local function Update(inputX)
            local rel = math.clamp((inputX - bg.AbsolutePosition.X) / bg.AbsoluteSize.X, 0, 1)
            f.Size = UDim2.new(rel, 0, 1, 0)
            onChange(rel)
        end

        bg.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                isDragging = true
                Update(input.Position.X)
            end
        end)

        UIS.InputChanged:Connect(function(input)
            if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                Update(input.Position.X)
            end
        end)

        UIS.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                isDragging = false
            end
        end)
    end

    local function Refresh()
        local c = Color3.new(r, g, b)
        preview.BackgroundColor3 = c
        if callback then callback(c) end
    end

    CreateColorSlider(38, "R", r, function(v) r = v; Refresh() end)
    CreateColorSlider(68, "G", g, function(v) g = v; Refresh() end)
    CreateColorSlider(98, "B", b, function(v) b = v; Refresh() end)
end

-- ЛОГИКА ПОИСКА БЛИЖАЙШЕЙ ЦЕЛИ
-- ========================================================
local function GetClosestPlayer()
    local closestPlayer = nil
    local shortestDist = math.huge
    local viewportCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            local head = p.Character:FindFirstChild("Head") or p.Character:FindFirstChild("HumanoidRootPart")
            if head then
                local screenPos, onScreen = Camera:WorldToViewportPoint(head.Position)
                if onScreen then
                    local screenVec = Vector2.new(screenPos.X, screenPos.Y)
                    local distToCenter = (screenVec - viewportCenter).Magnitude

                    if Settings.EnableFOV then
                        if distToCenter <= Settings.FOVSize and distToCenter < shortestDist then
                            shortestDist = distToCenter
                            closestPlayer = p
                        end
                    else
                        if distToCenter < shortestDist then
                            shortestDist = distToCenter
                            closestPlayer = p
                        end
                    end
                end
            end
        end
    end
    return closestPlayer
end

            if head then
                Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, head.Position)
            end
        end
    end
end)

local rawmeta = getrawmetatable and getrawmetatable(game)
if rawmeta and setreadonly and hookmetamethod then
    local oldIndex = rawmeta.__index
    setreadonly(rawmeta, false)

    rawmeta.__index = newcclosure(function(self, idx)
        if not checkcaller() and Settings.SilentAim and self == mouse then
            if idx == "Hit" or idx == "Target" then
                local target = GetClosestPlayer()
                if target and target.Character then
                    local head = target.Character:FindFirstChild("Head") or target.Character:FindFirstChild("HumanoidRootPart")
                    if head then
                        if idx == "Hit" then
                            return head.CFrame
                        elseif idx == "Target" then
                            return head
                        end
                    end
                end
            end
        end
        return oldIndex(self, idx)
    end)
    setreadonly(rawmeta, true)
end

local ESPCache = {}

local function RemoveESP(p)
    if ESPCache[p] then
        if ESPCache[p].Highlight then ESPCache[p].Highlight:Destroy() end
        if ESPCache[p].Billboard then ESPCache[p].Billboard:Destroy() end
        ESPCache[p] = nil
    end
end

local function CreateESP(p)
    if p == player then return end

    local function SetupCharacter(char)
        if not char then return end
        local hrp = char:WaitForChild("HumanoidRootPart", 5)
        local hum = char:WaitForChild("Humanoid", 5)
        if not hrp or not hum then return end

        RemoveESP(p)

        -- Highlight Подсветка
        local hl = Instance.new("Highlight")
        hl.Name = "SENSE_Highlight"
        hl.Adornee = char
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 0.5
        hl.OutlineTransparency = 0
        hl.Enabled = false
        hl.Parent = Gui

        -- Billboard UI (Рамка Box + HealthBar)
        local bb = Instance.new("BillboardGui")
        bb.Name = "SENSE_ESP"
        bb.Adornee = hrp
        bb.Size = UDim2.fromOffset(45, 60)
        bb.AlwaysOnTop = true
        bb.ClipsDescendants = false
        bb.Parent = Gui

        local boxFrame = Instance.new("Frame")
        boxFrame.Size = UDim2.new(1, 0, 1, 0)
        boxFrame.BackgroundTransparency = 1
        boxFrame.Parent = bb

        local boxStroke = Instance.new("UIStroke")
        boxStroke.Color = Settings.ESPColor
        boxStroke.Thickness = 1.5
        boxStroke.Parent = boxFrame

        local healthBg = Instance.new("Frame")
        healthBg.Size = UDim2.new(0, 4, 1, 0)
        healthBg.Position = UDim2.new(0, -8, 0, 0)
        healthBg.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
        healthBg.BorderSizePixel = 0
        healthBg.Parent = bb

        local healthFill = Instance.new("Frame")
        healthFill.Size = UDim2.new(1, 0, 1, 0)
        healthFill.Position = UDim2.new(0, 0, 0, 0)
        healthFill.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
        healthFill.BorderSizePixel = 0
        healthFill.Parent = healthBg

        ESPCache[p] = {
            Highlight = hl,
            Billboard = bb,
            Box = boxFrame,
            BoxStroke = boxStroke,
            HealthBg = healthBg,
            HealthFill = healthFill,
            Char = char,
            Hum = hum
        }
    end

    if p.Character then SetupCharacter(p.Character) end
    p.CharacterAdded:Connect(SetupCharacter)
end

            esp.HealthFill.Position = UDim2.new(0, 0, 1 - hpPercent, 0)
            esp.HealthFill.BackgroundColor3 = Color3.fromRGB(255 * (1 - hpPercent), 255 * hpPercent, 50)
        else
            if esp.Highlight then esp.Highlight.Enabled = false end
            if esp.Billboard then esp.Billboard.Enabled = false end
        end
    end
